cmake_minimum_required(VERSION 3.21)
project(prisma3d VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

option(PRISMA3D_BUILD_TESTS "Build the tests" OFF)
option(PRISMA3D_BUILD_TOOLS "Build the tools" ON)
option(PRISMA3D_BUILD_EDITOR "Build the editor" ON)
option(PRISMA3D_USE_CCACHE "Compile through ccache when it is installed" ON)
option(PRISMA3D_USE_MOLD "Link with mold when it is installed" ON)
set(PRISMA3D_BIN_DIR "${CMAKE_SOURCE_DIR}/bin" CACHE PATH "Folder for the editor, the runner and the tools")
option(PRISMA3D_WERROR "Treat compiler warnings as errors in the engine, editor, runner, tools and tests" ON)

set(PRISMA3D_WARNINGS -Wall -Wextra)
option(PRISMA3D_GC_SECTIONS "Drop unreferenced functions and data from the executables" ON)
if(PRISMA3D_GC_SECTIONS AND NOT MSVC)
  set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -ffunction-sections -fdata-sections")
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -ffunction-sections -fdata-sections")
  set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -Wl,--gc-sections")
endif()
if(PRISMA3D_WERROR)
  list(APPEND PRISMA3D_WARNINGS -Werror)
endif()
if(WIN32)
  add_compile_definitions(NOMINMAX)
endif()
if(MINGW)
  add_compile_definitions(__USE_MINGW_ANSI_STDIO=1)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  list(APPEND PRISMA3D_WARNINGS -Wno-maybe-uninitialized
    $<$<COMPILE_LANGUAGE:CXX>:-Wno-dangling-reference>
    $<$<COMPILE_LANGUAGE:CXX>:-Wno-array-bounds>
    $<$<COMPILE_LANGUAGE:CXX>:-Wno-stringop-overread>)
endif()

if(PRISMA3D_USE_CCACHE AND NOT CMAKE_C_COMPILER_LAUNCHER AND NOT CMAKE_CXX_COMPILER_LAUNCHER)
  find_program(PRISMA3D_CCACHE ccache)
  if(PRISMA3D_CCACHE)
    set(CMAKE_C_COMPILER_LAUNCHER ${PRISMA3D_CCACHE})
    set(CMAKE_CXX_COMPILER_LAUNCHER ${PRISMA3D_CCACHE})
  endif()
endif()
if(PRISMA3D_USE_MOLD AND NOT MSVC)
  find_program(PRISMA3D_MOLD mold)
  if(PRISMA3D_MOLD)
    add_link_options(-fuse-ld=mold)
  endif()
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

add_library(ct INTERFACE)
target_include_directories(ct INTERFACE external/containers/include)

set(MATHC_NATIVE_ARCH OFF CACHE BOOL "" FORCE)
set(MATHC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MATHC_BUILD_BENCH OFF CACHE BOOL "" FORCE)
add_subdirectory(external/math EXCLUDE_FROM_ALL)

add_subdirectory(external/Kinetix/core EXCLUDE_FROM_ALL)
add_subdirectory(external/Kinetix/3d EXCLUDE_FROM_ALL)
add_subdirectory(external/Kinetix/ai EXCLUDE_FROM_ALL)
if(NOT MSVC)
  target_compile_options(kinetix-core PRIVATE -fno-exceptions -fno-rtti)
  target_compile_options(kinetix-3d PRIVATE -fno-exceptions -fno-rtti)
  target_compile_options(kinetix-ai PRIVATE -fno-exceptions -fno-rtti)
endif()

set(PRISMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENV{SOURCE_DATE_EPOCH} 0)
add_subdirectory(external/prisma)
set(PRISMA_ROOT_DIR ${PROJECT_SOURCE_DIR}/external/prisma)
set(PRISMA_SHADER_INCLUDE_DIR ${PROJECT_SOURCE_DIR}/libprisma3d/shaders)
include(${PRISMA_ROOT_DIR}/cmake/PrismaShaders.cmake)
if(NOT PRISMA_SHADER_TOOLS)
  message(FATAL_ERROR "prisma3d needs glslang and spirv-cross to build its shaders. "
                      "If spirv-cross is missing, run: git submodule update --init --recursive")
endif()

set(PRISMA_SHADER_ESSL OFF CACHE BOOL "" FORCE)
set(PLATFORM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PLATFORM_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(external/zen_plataform EXCLUDE_FROM_ALL)

set(ZEN_HOST_OUTPUT ON CACHE BOOL "" FORCE)
set(ZEN_MODULE_NUMPY OFF CACHE BOOL "")
set(ZEN_MODULE_HTTP OFF CACHE BOOL "")
set(ZEN_MODULE_NET OFF CACHE BOOL "")
add_subdirectory(external/zenpy/libzen EXCLUDE_FROM_ALL)
target_include_directories(zen_static BEFORE PRIVATE ${PROJECT_SOURCE_DIR}/external/containers/include)

set(IGUI_CONTAINERS_DIR ${PROJECT_SOURCE_DIR}/external/containers CACHE PATH "" FORCE)
set(IGUI_FONT_INPUT ${PROJECT_SOURCE_DIR}/libprisma3d/third_party/fonts/Roboto-Regular.ttf CACHE FILEPATH "" FORCE)
set(IGUI_BUILD_FONT ON CACHE BOOL "" FORCE)
set(IGUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(IGUI_BUILD_RETAINED ${PRISMA3D_BUILD_EDITOR} CACHE BOOL "" FORCE)
set(IGUI_BUILD_LEGACY OFF CACHE BOOL "" FORCE)
set(IGUI_BUILD_RAYLIB_BACKEND OFF CACHE BOOL "" FORCE)
add_subdirectory(external/iGUI EXCLUDE_FROM_ALL)

set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
set(ENABLE_PCH OFF CACHE BOOL "" FORCE)
set(BUILD_EXTERNAL OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(GLSLANG_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(external/glslang EXCLUDE_FROM_ALL)
add_subdirectory(libprisma3d)

if(PRISMA3D_BUILD_TOOLS OR PRISMA3D_BUILD_TESTS)
  add_subdirectory(tools)
endif()

if(PRISMA3D_BUILD_EDITOR)
  add_subdirectory(editor)
  add_subdirectory(runner)
endif()

if(PRISMA3D_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
