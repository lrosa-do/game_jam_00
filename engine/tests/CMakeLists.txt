function(prisma3d_test name)
  add_executable(test_${name} test_${name}.cpp)
  target_link_libraries(test_${name} PRIVATE prisma3d)
  target_compile_definitions(test_${name} PRIVATE PRISMA3D_TEST_DIR="${CMAKE_CURRENT_BINARY_DIR}")
  if(NOT MSVC)
    target_compile_options(test_${name} PRIVATE ${PRISMA3D_WARNINGS} -fno-exceptions -fno-rtti)
  endif()
  add_test(NAME ${name} COMMAND test_${name})
endfunction()
